# Opt-in ML extras for in-process CPU features (input-box autocomplete and the
# slop, emotion, POV, and markup classifiers). Base Orb runs without these.
#   pip install -r requirements-ml.txt
# Then use Settings → Local ML to download the models into backend/data/models/,
# or set ORB_AUTOCOMPLETE_MODEL to an existing typeahead GGUF's path.
#
# The Prose Rewriter runs a child llama-server and needs none of these.
llama-cpp-python>=0.3.0

# The built-in Spark-TTS voice cloner (runtime="onnx"): the speaker encoder and
# the bicodec decoder are ONNX graphs, so onnxruntime is all it needs at run
# time. The `onnx` package — another 73 MB — only BUILDS graphs and lives in
# requirements-dev.txt with scripts/export_spark_speaker_encoder.py.
onnxruntime>=1.17
# Optional, and only for enrolling a voice: the stdlib reads PCM WAV, this adds
# FLAC/OGG and the WAV encodings it rejects, and an ffmpeg on PATH covers
# MP3/M4A. Enrollment degrades to WAV-only without it rather than failing.
soundfile>=0.12
